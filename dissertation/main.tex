%========================================================================================
% FIX KAOBOOK FLOAT(S) LOST WITH FIRA SANS LIGHT
%========================================================================================
% Fira Sans Light produces slightly different page breaks than EB Garamond, causing
% sidenote floats (marginpars) to queue up and get lost at \clearpage.
% Fix: globally override \marginnote to always use the non-float \oldmarginnote path,
% preventing sidenotes from being deferred to pages where they can't fit.
% Visual effect: notes appear exactly at their anchor position (no page-float offset).
\makeatletter
\AtBeginDocument{%
    \RenewDocumentCommand\marginnote{ o m }{%
        \IfNoValueOrEmptyTF{#1}{%
            \oldmarginnote{#2}%
        }{%
            \oldmarginnote{#2}[\kao@if@nblskip{#1}{\@cdr#1\@nil\baselineskip}{#1}]%
        }%
    }%
}
\makeatother

%========================================================================================
% FIX KAOBOOK CAPTION RECURSION
%========================================================================================
% The kao.sty caption hooks cause infinite recursion with some packages.
% Solution: Patch the figure and table environments to reset \caption
\makeatletter
\AtBeginDocument{%
    % Save the original caption command (before kao.sty hooks redefine it)
    \let\kaociOrigCaption\caption%
    % Patch the figure begin to restore original caption
    \pretocmd{\figure}{\let\caption\kaociOrigCaption}{}{}%
    \pretocmd{\table}{\let\caption\kaociOrigCaption}{}{}%
}
\makeatother
